        function calculateSchedule() {

            const now = new Date(); const dayOfWeek = now.getDay();

            let orderDateObj = new Date(now); let deliveryDateObj = new Date(now);

            if (selectedBranchCode === 'BBB1') {

                let diffToFri = (5 - dayOfWeek + 7) % 7;

                orderDateObj.setDate(now.getDate() + (diffToFri === 0 ? 0 : diffToFri));

                deliveryDateObj = new Date(orderDateObj); deliveryDateObj.setDate(orderDateObj.getDate() + 3);

            } else {

                let diffToTue = (2 - dayOfWeek + 7) % 7;

                orderDateObj.setDate(now.getDate() + (diffToTue === 0 ? 0 : diffToTue));

                deliveryDateObj = new Date(orderDateObj); deliveryDateObj.setDate(orderDateObj.getDate() + 2);

            }

            const options = { year: 'numeric', month: 'short', day: 'numeric' };

            return { orderDate: orderDateObj.toLocaleDateString('th-TH', options), deliveryDate: deliveryDateObj.toLocaleDateString('th-TH', options) };

        }

            });

                container.innerHTML = `<div class="p-4 text-center text-gray-500 bg-gray-50 rounded-xl text-xs">${t.no_order_needed}</div>`;

                    <div class="bg-amber-50 border border-amber-200 p-2.5 rounded-xl flex items-center gap-2.5">

                        </div>

                        <div class="text-right">

        async function submitData() {

            const TARGET_URL = selectedBranchCode === 'MKP' ? WEB_APP_URL_MKP : WEB_APP_URL_BBB1;

            if (TARGET_URL === "https://script.google.com/macros/s/AKfycbyt311GDUEy70aYT4DFal_R8-UssAkKFV7prm4zndUZUXxqfUZTWIMytsk3hMLujn5SNA/exec" || TARGET_URL === "https://script.google.com/macros/s/AKfycby-cjM33qspevoBdrFCHWMeC1mLkN4upfd17X_pDEmBu-9cazJeuQajYYqi-rLrtPpftw/exec") {

                alert("กรุณาใส่ URL ของ Google Apps Script ในโค้ดก่อนใช้งานจริง");

                return;

            }


            const schedule = calculateSchedule();

            let stockItemsList = []; let orderItemsList = [];

            categories.forEach(cat => {

                cat.items.forEach(item => {

                    const val = stockValues[item.code] || { pack: 0, opened: 0 };

                    stockItemsList.push({ code: item.code, name: item.name, pack: val.pack, opened: val.opened });

                    if (workMode === 'order') {

                        const target = selectedBranchCode === 'MKP' ? item.mkpTarget : item.bbb1Target;

                        const orderQty = Math.max(0, target - val.pack);

                        if (orderQty > 0) orderItemsList.push({ code: item.code, name: item.name, orderQty: orderQty, unit: item.unit });

                    }

                });

            });


            try {

                await fetch(TARGET_URL, {

                    method: "POST", mode: "no-cors", headers: { "Content-Type": "application/json" },

                    body: JSON.stringify({ 

                        action: "saveStock", branch: selectedBranchName, mode: workMode,

                        checkDate: new Date().toLocaleDateString('th-TH'), orderDate: schedule.orderDate, deliveryDate: schedule.deliveryDate,

                        items: stockItemsList 

                    })

                });


                if (workMode === 'order' && orderItemsList.length > 0) {

                    await fetch(TARGET_URL, {

                        method: "POST", mode: "no-cors", headers: { "Content-Type": "application/json" },

                        body: JSON.stringify({ 

                            action: "saveOrder", branch: selectedBranchName, 

                            orderDate: schedule.orderDate, deliveryDate: schedule.deliveryDate, orders: orderItemsList 

                        })

                    });

                }

                alert(translations[currentLang].success_msg);

                location.reload();

            } catch (error) {

                console.error("Error:", error);

                alert("ระบบบันทึกข้อมูลเรียบร้อยแล้ว");

                location.reload();

            }

        }
